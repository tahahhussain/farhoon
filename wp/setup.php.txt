<?php
// Imports the Farhoon Elementor templates (Arabic and English), builds the pages and the site CSS.
// Run with: wp eval-file <this file> --user=<admin>. Safe to run again: pages are updated, not duplicated.
if (!defined('ABSPATH') || !class_exists('\Elementor\Plugin')) { exit("needs WordPress with Elementor\n"); }
$base = 'https://tahahhussain.github.io/farhoon/wp/';
$pages = [ // file, title, path
  ['farhoon-ar-1-home', 'الرئيسية', 'home'],
  ['farhoon-ar-2-about', 'من نحن', 'about'],
  ['farhoon-ar-3-rafiq', 'رفيق الطاعة', 'rafiq'],
  ['farhoon-ar-4-games', 'ألعابنا', 'games'],
  ['farhoon-ar-5-contact', 'تواصل معنا', 'contact'],
  ['farhoon-ar-6-rafiq-privacy', 'سياسة الخصوصية', 'rafiq/privacy'],
  ['farhoon-en-1-home', 'Farhoon', 'en'],
  ['farhoon-en-2-about', 'About', 'en/about'],
  ['farhoon-en-3-rafiq', 'Rafiq Al-Taa', 'en/rafiq'],
  ['farhoon-en-4-games', 'Games', 'en/games'],
  ['farhoon-en-5-contact', 'Contact', 'en/contact'],
  ['farhoon-en-6-rafiq-privacy', 'Privacy policy', 'en/rafiq/privacy'],
];
// Older copies of these templates in the library are replaced by the new ones.
foreach (get_posts(['post_type' => 'elementor_library', 'numberposts' => -1, 'post_status' => 'any']) as $old) {
  if (preg_match('/^(فرحون|رفيق الطاعة|Farhoon|Rafiq Al-Taa) - /u', $old->post_title)) wp_delete_post($old->ID, true);
}
$source = \Elementor\Plugin::$instance->templates_manager->get_source('local');
$ids = [];
foreach ($pages as [$file, $title, $path]) {
  $res = wp_remote_get($base . $file . '.json', ['timeout' => 60]);
  if (is_wp_error($res) || wp_remote_retrieve_response_code($res) !== 200) { echo "FAILED download $file\n"; continue; }
  $tmp = wp_tempnam($file . '.json');
  file_put_contents($tmp, wp_remote_retrieve_body($res));
  $items = $source->import_template($file . '.json', $tmp);
  @unlink($tmp);
  if (is_wp_error($items) || empty($items[0]['template_id'])) { echo "FAILED import $file\n"; continue; }
  $tpl = $items[0]['template_id'];
  $slug = basename($path);
  $parent_path = dirname($path) === '.' ? '' : dirname($path);
  $parent_id = $parent_path ? ($ids[$parent_path] ?? (get_page_by_path($parent_path)->ID ?? 0)) : 0;
  $existing = get_page_by_path($path, OBJECT, 'page');
  $id = $existing ? $existing->ID : wp_insert_post(['post_title' => $title, 'post_name' => $slug, 'post_status' => 'publish', 'post_type' => 'page', 'post_parent' => $parent_id]);
  if ($existing) wp_update_post(['ID' => $id, 'post_title' => $title]);
  foreach (['_elementor_data', '_elementor_page_settings', '_elementor_version'] as $k) {
    update_post_meta($id, $k, wp_slash(get_post_meta($tpl, $k, true)));
  }
  update_post_meta($id, '_elementor_edit_mode', 'builder');
  update_post_meta($id, '_wp_page_template', 'elementor_canvas');
  $ids[$path] = $id;
  echo "OK $path page=$id " . get_permalink($id) . "\n";
}
if (!empty($ids['home'])) { update_option('show_on_front', 'page'); update_option('page_on_front', $ids['home']); }
// Site-wide CSS (sticky header, menu, English pages) goes in Appearance > Customize > Additional CSS.
$css = wp_remote_retrieve_body(wp_remote_get($base . 'farhoon-site.css', ['timeout' => 60]));
if ($css) { wp_update_custom_css_post($css, ['stylesheet' => get_stylesheet()]); echo "css saved\n"; }
// The app links to /rafiq/privacy.html; send it to the WordPress page.
$ht = ABSPATH . '.htaccess';
$cur = file_exists($ht) ? file_get_contents($ht) : '';
if (strpos($cur, '/rafiq/privacy.html') === false) { file_put_contents($ht, "# Farhoon: app privacy link\nRedirect 301 /rafiq/privacy.html " . home_url('/rafiq/privacy/') . "\n" . $cur); echo "redirect added\n"; }
\Elementor\Plugin::$instance->files_manager->clear_cache();
echo "DONE\n";
